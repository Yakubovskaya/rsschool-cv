const duplicateEncode = (word) => {
  const sameLetters = {};
  const letters = word.toLowerCase().split('');

  letters.forEach(
    (el) => sameLetters[el] = (sameLetters[el] || 0) + 1);

  return letters
    .map((el) => sameLetters[el] === 1 ? '(' : ')')
    .join('');
};
